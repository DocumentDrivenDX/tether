"""Does a per-call Authorization header (explicit identity, no ambient context variable) override the client default, under concurrency?"""
import asyncio, json
import httpx
from openai import AsyncOpenAI
from pydantic_ai.direct import model_request
from pydantic_ai.messages import ModelRequest
from pydantic_ai.models.openai import OpenAIChatModel
from pydantic_ai.providers.openai import OpenAIProvider
from pydantic_ai.settings import ModelSettings

seen = []
def handler(request):
    seen.append(request.headers.get("authorization"))
    return httpx.Response(200, json={"id": "c", "object": "chat.completion", "created": 1, "model": "m",
        "choices": [{"index": 0, "finish_reason": "stop", "message": {"role": "assistant", "content": "hi"}}],
        "usage": {"prompt_tokens": 3, "completion_tokens": 1, "total_tokens": 4}},
        headers={"x-request-id": "req-for-" + (request.headers.get("authorization") or "none").removeprefix("Bearer ")})

http = httpx.AsyncClient(transport=httpx.MockTransport(handler))
client = AsyncOpenAI(base_url="https://gateway.example/v1", api_key="client-default-key", http_client=http, max_retries=0)
model = OpenAIChatModel("tier-balanced", provider=OpenAIProvider(openai_client=client))

class Credential:                                   # explicit, per-actor, async (a token that may need refreshing)
    def __init__(self, token): self._t = token
    async def authorization(self): return f"Bearer {self._t}"

async def call(i):
    cred = Credential(f"user-{i}")
    settings = ModelSettings(extra_headers={"Authorization": await cred.authorization()})
    await model_request(model, [ModelRequest.user_text_prompt("hi")], model_settings=settings)

async def main():
    await model_request(model, [ModelRequest.user_text_prompt("hi")])              # no per-call header: the client default
    await asyncio.gather(*(call(i) for i in range(5)))
    print("default call used:", seen[0])
    print("per-call headers on the wire:", sorted(seen[1:]))
    print("client default never used for per-call requests:", all("client-default-key" not in s for s in seen[1:]))
asyncio.run(main())
