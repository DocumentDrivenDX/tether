OPERATOR_PATH = Path('/private/tmp/tether-run-qualification-ram-contextual27.py')
HOST = 'https://workspace.fabricated.example'
CLIENT_ID = 'fabricated-client-id'
SECRET = 'fabricated-created-secret'
PROFILE = 'fabricated-profile'
AUTHORIZATION = 'Bearer ' + 'header.' + 'payload.' + 'signature'


def _operator() -> ModuleType:
    spec = importlib.util.spec_from_file_location('qualification_ram_contextual27', OPERATOR_PATH)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def _authorization(expiry: float) -> str:
    header = base64.urlsafe_b64encode(b'{"alg":"none"}').decode().rstrip('=')
    payload = base64.urlsafe_b64encode(
        json.dumps({'exp': expiry}, separators=(',', ':')).encode()
    ).decode().rstrip('=')
    return f'Bearer {header}.{payload}.fabricated-signature'


def _config(path: Path, *, host: str = HOST, client_id: str = CLIENT_ID) -> Path:
    path.write_text(
        f'[{PROFILE}]\nhost = {host}\nclient_id = {client_id}\n', encoding='utf-8'
    )
    return path


def _factory(
    delegate: Any,
    path: Path,
    *,
    secret_reader=lambda: SECRET,
):
    return _operator().ContextualCredentialFactory(
        delegate,
        config_path=path,
        profile=PROFILE,
        host=HOST,
        client_id=CLIENT_ID,
        secret=SECRET,
        secret_reader=secret_reader,
    )


def test_exact_rejection_translates_only_after_prior_future_jwt_mint(tmp_path: Path) -> None:
    config_path = _config(tmp_path / 'service.ini')
    calls = 0
    first = credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))
    rejection = RuntimeError('identity_credential_rejected')

    async def delegate():
        nonlocal calls
        calls += 1
        if calls == 1:
            return first
        raise rejection

    factory = _factory(delegate, config_path)
    assert SECRET not in repr(factory)
    assert asyncio.run(factory()) is first
    with pytest.raises(credential_module.CredentialRefreshError) as caught:
        asyncio.run(factory())
    assert caught.value.args == ('identity_credential_rejected_after_prior_fresh_mint',)
    assert caught.value.retryable is True
    assert factory.retryable_rejections == 1
    assert calls == 2


def test_initial_or_expired_rejection_is_preserved_unchanged(tmp_path: Path) -> None:
    config_path = _config(tmp_path / 'service.ini')
    initial_error = RuntimeError('identity_credential_rejected')

    async def initial_failure():

    initial = _factory(initial_failure, config_path)
    with pytest.raises(RuntimeError) as caught_initial:

    calls = 0

    async def expired_failure():
        nonlocal calls
        calls += 1
        return expired if calls == 1 else (_ for _ in ()).throw(expired_error)

    after_expiry = _factory(expired_failure, config_path)
    assert asyncio.run(after_expiry()) is expired
    with pytest.raises(RuntimeError) as caught_expired:
        asyncio.run(after_expiry())
    assert caught_expired.value is expired_error
    assert after_expiry.retryable_rejections == 0

def test_only_exact_runtime_error_is_translated_and_direct_call_propagates(
    tmp_path: Path,
) -> None:
    config_path = _config(tmp_path / 'service.ini')
    good = credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))
    calls = 0

    class RuntimeErrorSubclass(RuntimeError):
        pass

    typed_error = credential_module.CredentialRefreshError('already typed', retryable=True)
    subclass_error = RuntimeErrorSubclass('identity_credential_rejected')
    wrong_args_error = RuntimeError('identity_credential_rejected', 'extra')
    unrelated_error = ValueError('unrelated')
    cases = [typed_error, subclass_error, wrong_args_error, unrelated_error]

    async def delegate():
        nonlocal calls
        calls += 1
        if calls == 1:
            return good
        raise cases.pop(0)

    factory = _factory(delegate, config_path)
    assert asyncio.run(factory()) is good
    for expected in (typed_error, subclass_error, wrong_args_error, unrelated_error):
        with pytest.raises(type(expected)) as caught:
            asyncio.run(factory())
        assert caught.value is expected
    assert factory.retryable_rejections == 0

    direct_calls = 0
    rejected = RuntimeError('identity_credential_rejected')

    async def direct_delegate():
        nonlocal direct_calls
        direct_calls += 1
        return good if direct_calls == 1 else (_ for _ in ()).throw(rejected)

    direct = _factory(direct_delegate, config_path)
    assert asyncio.run(direct()) is good
    with pytest.raises(credential_module.CredentialRefreshError):
        asyncio.run(direct())
    assert direct_calls == 2
    assert direct.retryable_rejections == 1

def test_cancellation_propagates_even_if_context_drifts_during_call(tmp_path: Path) -> None:
    config_path = _config(tmp_path / 'service.ini')
    started = asyncio.Event()

    async def blocked_delegate():
        started.set()
        await asyncio.Event().wait()

    factory = _factory(blocked_delegate, config_path)

    async def invoke_and_cancel():
        task = asyncio.create_task(factory())
        await started.wait()
        config_path.write_text(
            f'[{PROFILE}]\nhost = {HOST}\nclient_id = changed-client\n', encoding='utf-8'
        )
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task

    asyncio.run(invoke_and_cancel())
    assert factory.retryable_rejections == 0

    rejection = RuntimeError('identity_credential_rejected')
    target = tmp_path / 'service2.ini'
    _config(target)

    async def drift_during_failure():
        target.write_text(
            f'[{PROFILE}]\nhost = {HOST}\nclient_id = changed-client\n', encoding='utf-8'
        )
        raise rejection

    changed = _factory(drift_during_failure, target)
    with pytest.raises(RuntimeError, match='contextual_credential_config_changed'):
        asyncio.run(changed())
    assert changed.retryable_rejections == 0


def test_config_and_secret_are_rechecked_after_successful_call(tmp_path: Path) -> None:
    config_path = _config(tmp_path / 'service.ini')
    calls_before = 0

    async def should_not_run():
        nonlocal calls_before
        calls_before += 1
        return credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))

    before = _factory(should_not_run, config_path)
    config_path.write_text(
        f'[{PROFILE}]\nhost = {HOST}\nclient_id = changed-client\n', encoding='utf-8'
    )
    with pytest.raises(RuntimeError, match='contextual_credential_config_changed'):
        asyncio.run(before())
    assert calls_before == 0

    config_path = _config(tmp_path / 'service.ini')
    changed_secret = False
    credential = credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))

    async def mutate_context_and_succeed():
        nonlocal changed_secret
        config_path.write_text(
            f'[{PROFILE}]\nhost = {HOST}\nclient_id = changed-client\n', encoding='utf-8'
        )
        changed_secret = True
        return credential

    factory = _factory(
        mutate_context_and_succeed,
        config_path,
        secret_reader=lambda: 'different-secret' if changed_secret else SECRET,
    )
    with pytest.raises(RuntimeError, match='contextual_credential_config_changed'):
        asyncio.run(factory())
    assert factory._last_credential is None

def test_restoration_loop_retries_then_uses_new_credential_for_real_readiness_checks(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    import httpx

    config_path = _config(tmp_path / 'service.ini')
    old = credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))
    new = credential_module.FreshCredential(HOST, _authorization(time.time() + 7200))
    calls = 0

    async def delegate():
        nonlocal calls
        calls += 1
        if calls == 1:
            return old
        if calls == 2:
            raise RuntimeError('identity_credential_rejected')
        return new

    factory = _factory(delegate, config_path)
    assert asyncio.run(factory()) is old
    seen_authorizations: list[str] = []

    def handler(request: httpx.Request) -> httpx.Response:
        seen_authorizations.append(request.headers.get('authorization', ''))
        return httpx.Response(200, json={'id': 'fabricated-id', 'active': True})

    transport = httpx.MockTransport(handler)
    monkeypatch.setattr(
        probe,
        '_client',
        lambda _host: httpx.AsyncClient(
            base_url=HOST, timeout=httpx.Timeout(10), trust_env=False, verify=True, transport=transport
        ),
    )
    protected_authorizations: list[str] = []

    def protected_status_for(token: str):
        async def protected() -> int:
            protected_authorizations.append(token)
            return 200

        return protected

    result = asyncio.run(
        probe.fresh_workspace_access(factory, HOST, 'fabricated-id', protected_status_for)
    )
    assert result is new
    assert calls == 3
    assert factory.retryable_rejections == 1
    assert seen_authorizations == [new.authorization]
    assert protected_authorizations == [new.authorization]
    assert old.authorization not in seen_authorizations
    assert old.authorization not in protected_authorizations


def test_restoration_loop_retries_contextual_rejection_until_original_deadline(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    config_path = _config(tmp_path / 'service.ini')
    future = credential_module.FreshCredential(HOST, _authorization(time.time() + 3600))
    calls = 0

    async def delegate():
        nonlocal calls
        calls += 1
        return future if calls == 1 else (_ for _ in ()).throw(
            RuntimeError('identity_credential_rejected')
        )

    factory = _factory(delegate, config_path)
    assert asyncio.run(factory()) is future
    monkeypatch.setattr(probe, '_DEADLINE_SECONDS', 0.04)
    monkeypatch.setattr(probe, '_RESTORE_AUTH_MIN_REMAINING_SECONDS', 0.0)
    monkeypatch.setattr(probe, '_POLL_INTERVAL_SECONDS', 0.001)
    protected_called = False

    def protected_status_for(_token: str):
        async def protected():
            nonlocal protected_called
            protected_called = True
            return 200

        return protected

    with pytest.raises(RuntimeError, match='identity_workspace_transition_timeout'):
        asyncio.run(probe.fresh_workspace_access(factory, HOST, 'fabricated-id', protected_status_for))
    assert calls > 2
    assert factory.retryable_rejections > 0
    assert not protected_called
    assert factory._last_credential is future
