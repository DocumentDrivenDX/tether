MODE='instrumented-qualification-only'
_FAILURE_STATE:dict[str,Any]={}
_WRAPPED_FAILURE_PHASES={
    'reopen_remote_client',
    'verify_recovery_barrier',
    'browser_recovery',
}
_WRAPPED_FAILURE_CODES={
    'qualification_restart_readiness_timeout',
    'live_recovered_session_read_failed',
    'live_unknown_effect_recovery_not_visible',
    'live_unknown_effect_barrier_not_enforced',
    'live_browser_recovery_terminal_status_invalid',
    'live_browser_case_failed',
    'live_browser_rendezvous_invalid',
    'live_browser_exited_before_rendezvous',
    'RemoteProblem',
    'RemoteTransportError',
    'TimeoutError',
}

def _set_phase(value:str)->None:
    if value in {
        'startup','input_validation','candidate_verification','workspace_snapshot',
        'identity_setup','public_qualifier','worker_probe','platform_stop',
        'wait_stopped','wait_running','stopped_snapshot','observer_connect','observer_query','observer_evaluate',
        'receipt_write','restart_start','restart_start_returned','restart_recovery',
        'qualifier_returned','complete',
    }:
        _FAILURE_STATE['phase']=value

def _wrapped_live_failure(error:BaseException)->tuple[str,str]|None:
    if type(error) is not RuntimeError or len(error.args)!=1 or type(error.args[0]) is not str:
        return None
    message=error.args[0]
    for phase in _WRAPPED_FAILURE_PHASES:
        prefix=f'live_unknown_effect_{phase}_'
        if not message.startswith(prefix):
            continue
        code=message[len(prefix):]
        if code in _WRAPPED_FAILURE_CODES:
            return phase,code
        return None
    return None

def _write_private_stop_binding(path:Path)->None:
    names=(
        'stop_correlation_id','deployment_id','source_code_path','worker_invocation_id',
        'base_candidate_sha256','overlay_snapshot_sha256','candidate_modules_byte_identical',
        'stop_command_returned','stop_state_observed','stop_state','snapshot_deployment_id',
        'snapshot_source_code_path','active_snapshot_matched','restart_running_observed',
    )
    record={
        'schema_version':1,'stop_attempted':_FAILURE_STATE.get('stop_attempted') is True,
        **{name:_FAILURE_STATE.get(name) for name in names if name in _FAILURE_STATE},
    }
    path.parent.mkdir(mode=0o700,parents=True,exist_ok=True)
    temporary=path.with_suffix('.tmp')
    temporary.write_text(json.dumps(record,sort_keys=True,separators=(',',':'))+'\n')
    os.chmod(temporary,0o600)
    temporary.replace(path)

def _safe_failure(error:BaseException)->dict[str,object]:
    classes={TimeoutError:'timeout',OSError:'io',RuntimeError:'runtime',ValueError:'value',
             KeyError:'key',TypeError:'type',ImportError:'import',AttributeError:'attribute',
             asyncio.CancelledError:'cancelled'}
    category=next((label for kind,label in classes.items() if isinstance(error,kind)),'other')
    names={TimeoutError:'TimeoutError',OSError:'OSError',RuntimeError:'RuntimeError',
           ValueError:'ValueError',KeyError:'KeyError',TypeError:'TypeError',
           ImportError:'ImportError',AttributeError:'AttributeError',
           asyncio.CancelledError:'CancelledError'}
    exception_name=next((label for kind,label in names.items() if isinstance(error,kind)),'other')
    known_exception_names={'RemoteProblem','ApiFailure','DatabricksError','OperationFailed','HTTPStatusError'}
    if type(error).__name__ in known_exception_names:
        exception_name=type(error).__name__
    known_codes={
        'active_deployment_mismatch','app_name_missing','app_not_running',
        'base_candidate_digest_mismatch','candidate_bytes_changed','candidate_runtime_source_mismatch',
        'composition_module_ambiguous','databricks_wheel_ambiguous','deployment_reference_invalid',
        'durable_shutdown_row_not_qualified','immutable_deployment_mismatch','immutable_export_empty',
        'immutable_export_unavailable','immutable_snapshot_bytes_mismatch','instrumented_mode_required',
        'multiple_stop_commands_refused','observer_overlay_settings_mismatch',
        'observer_reader_identity_unavailable','observer_stop_prerequisite_missing',
        'platform_stop_state_not_observed','probe_nonce_invalid','project_root_unrecognized',
        'protected_probe_not_correlated','reader_unavailable','settings_invalid',
        'stopped_active_deployment_mismatch','stopped_snapshot_mismatch',
    }
    code='other'
    inner_phase=None
    wrapped=_wrapped_live_failure(error)
    if wrapped is not None:
        inner_phase,code=wrapped
    if type(error) is RuntimeError and len(error.args)==1 and type(error.args[0]) is str and error.args[0] in known_codes:
        code=error.args[0]
    allowed_files={
        'run_instrumented_qualification.py','qualify_live_apps.py',
        'qualify_live_apps_lifecycle.py','qualify_live_apps_transport.py',
        'read_observer.py','observer_launcher.py','app.py',
    }
    safe_frame:dict[str,object]={}
    tb=error.__traceback__
    while tb is not None:
        frame=tb.tb_frame
        name=Path(frame.f_code.co_filename).name
        if name in allowed_files:
            safe_frame={'failure_file':name,'failure_line':tb.tb_lineno,'failure_function':frame.f_code.co_name}
        tb=tb.tb_next
    failure_status=None
    error_fields=getattr(error,'__dict__',{})
    if isinstance(error_fields,dict):
        candidate_status=error_fields.get('status')
        if type(candidate_status) is int and 100<=candidate_status<=599:
            failure_status=candidate_status
        candidate_code=error_fields.get('code')
        known_error_codes={'REMOTE_REQUEST_FAILED','PERMISSION_DENIED','INVALID_TOKEN','UNAUTHENTICATED','INTERNAL','UNKNOWN'}
        if type(candidate_code) is str and candidate_code in known_error_codes:
            code=candidate_code
    result={'failure_category':category,'failure_exception':exception_name,'failure_code':code,'failure_status':failure_status,**safe_frame}
    if inner_phase is not None:
        result['failure_inner_phase']=inner_phase
    return result

def _failure_result(error:BaseException)->dict[str,object]:
    return {
        'schema_version':1,'mode':MODE,'status':'failed',**_safe_failure(error),**{
            key:_FAILURE_STATE.get(key) for key in (
                'phase','stop_attempted','stop_state_observed','row_query_attempted','row_found',
                'reader_status','protected_probe_observed','worker_started_observed',
                'stop_binding_record_written','stop_binding_sha256','restart_running_observed',
            )
        },
    }


def _digest_inventory(values:dict[str,str])->str:
    return hashlib.sha256(json.dumps(values,sort_keys=True,separators=(',',':')).encode()).hexdigest()

def _sha(path:Path)->str:
    return hashlib.sha256(path.read_bytes()).hexdigest()

def _status_value(value:Any)->Any:
    state=getattr(value,'state',None)
    return getattr(state,'value',state)

def _wheel_info(bundle:Path)->tuple[Path,str,str]:
    matches=list((bundle/'wheelhouse').glob('novirhiza_tether_databricks-*.whl'))
    if len(matches)!=1: raise RuntimeError('databricks_wheel_ambiguous')
    wheel=matches[0]
    with zipfile.ZipFile(wheel) as archive:
        names=[n for n in archive.namelist() if n.endswith('tether_databricks/composition.py')]
        if len(names)!=1: raise RuntimeError('composition_module_ambiguous')
        composition=hashlib.sha256(archive.read(names[0])).hexdigest()
    return wheel,_sha(wheel),composition

def _expected_app_manifest(base_text:str,values:dict[str,str])->str|None:
    lines=base_text.splitlines()
    try:
        command_start=lines.index('command:')
        env_start=next(i for i in range(command_start+1,len(lines)) if lines[i]=='env:')
    except (ValueError,StopIteration): return None
    if lines[command_start:env_start] != ['command:','  - "uv"','  - "run"','  - "--locked"','  - "python"','  - "src/app.py"']:
        return None
    names=[line.strip()[len('- name: '):] for line in lines[env_start+1:] if line.strip().startswith('- name: ')]
    required={'TETHER_DIAG_BASE_CANDIDATE_SHA256','TETHER_DIAG_SCHEMA','TETHER_DIAG_PROBE_NONCE','TETHER_DIAG_PROTECTED_PATH'}
    if len(names)!=len(set(names)) or required.intersection(names) or set(values)!=required:
        return None
    if any('\n' in value or '\r' in value for value in values.values()): return None
    command=['command:','  - "uv"','  - "run"','  - "--locked"','  - "python"','  - "src/observer_launcher.py"']
    expected=lines[:command_start]+command+lines[env_start:]
    for name in ('TETHER_DIAG_BASE_CANDIDATE_SHA256','TETHER_DIAG_SCHEMA','TETHER_DIAG_PROBE_NONCE','TETHER_DIAG_PROTECTED_PATH'):
        expected.extend((f'  - name: {name}',f'    value: {json.dumps(values[name])}'))
    return '\n'.join(expected)+'\n'

def _candidate_bytes_equal(base:dict[str,str],overlay:dict[str,str],base_app_text:str,overlay_app_text:str,diagnostic_values:dict[str,str],inventory_text:str)->bool:
    added={'src/observer_launcher.py','src/observer_metadata.py','src/read_observer.py','src/tether_h2c_probe.py','observer_payload_inventory.json'}
    if added.intersection(base) or set(overlay)!=set(base)|added: return False
    expected=_expected_app_manifest(base_app_text,diagnostic_values)
    if expected is None or overlay_app_text!=expected: return False
    try: inventory=json.loads(inventory_text)
    except (ValueError,TypeError): return False
    if inventory!={"schema_version":1,"files":sorted(overlay)}: return False
    return all(overlay.get(path)==digest for path,digest in base.items() if path!='app.yaml')

class _KeepTemporaryDirectory:
    def __init__(self,path:Path): self.path=path
    def __enter__(self)->str:
        self.path.mkdir(mode=0o700,parents=True,exist_ok=False)
        return str(self.path)
    def __exit__(self,*_args:object)->None: return None

async def _app_snapshot(workspace:Any,app_name:str,bundle:Path,expected_deployment_id:str,expected_source_path:str,*,require_stopped:bool)->tuple[bool,str,str]:
    from databricks.sdk.service.workspace import ExportFormat
    from selected_profiles_bundle import _local_inventory,_snapshot_inventory
    app=await asyncio.to_thread(workspace.apps.get,app_name)
    app_state=_status_value(app.app_status)
    compute_state=_status_value(app.compute_status)
    active=getattr(app,'active_deployment',None)
    active_id=getattr(active,'deployment_id',None)
    active_path=getattr(active,'source_code_path',None)
    if require_stopped:
        if app_state not in {'STOPPED','UNAVAILABLE'} or compute_state!='STOPPED':
            raise RuntimeError('platform_stop_state_not_observed')
        if active_id not in {None,expected_deployment_id} or active_path not in {None,expected_source_path}:
            raise RuntimeError('stopped_active_deployment_mismatch')
    elif app_state!='RUNNING' or compute_state!='ACTIVE' or active_id!=expected_deployment_id or active_path!=expected_source_path:
        raise RuntimeError('active_deployment_mismatch')
    deployment=await asyncio.to_thread(workspace.apps.get_deployment,app_name,expected_deployment_id)
    state=getattr(getattr(deployment,'status',None),'state',None)
    if getattr(deployment,'deployment_id',None)!=expected_deployment_id or getattr(state,'value',state)!='SUCCEEDED' or getattr(deployment,'source_code_path',None)!=expected_source_path:
        raise RuntimeError('immutable_deployment_mismatch')
    artifacts=getattr(deployment,'deployment_artifacts',None)
    export_path=getattr(artifacts,'source_code_path',None)
    if not isinstance(export_path,str) or not export_path: raise RuntimeError('immutable_export_unavailable')
    exported=await asyncio.to_thread(workspace.workspace.export,export_path,format=ExportFormat.AUTO)
    if not isinstance(exported.content,str) or not exported.content: raise RuntimeError('immutable_export_empty')
    import base64
    with zipfile.ZipFile(io.BytesIO(base64.b64decode(exported.content,validate=True))) as archive:
        remote=_snapshot_inventory(archive)
    local=await asyncio.to_thread(_local_inventory,bundle)
    if not local or remote!=local: raise RuntimeError('immutable_snapshot_bytes_mismatch')
    return True,_digest_inventory(local),app_state

    old_wait_app_state=lifecycle._wait_app_state
    async def observed_wait_app_state(profile:str,config_file:str|None,source:dict[str,str],app:str,accepted:set[tuple[str,str]],*,timeout_seconds:float)->None:
        waiting_for_stop=any(app_state in {'STOPPED','UNAVAILABLE'} and compute_state=='STOPPED' for app_state,compute_state in accepted)
        _set_phase('wait_stopped' if waiting_for_stop else 'wait_running')
        await old_wait_app_state(profile,config_file,source,app,accepted,timeout_seconds=timeout_seconds)
        if waiting_for_stop:
            _FAILURE_STATE['stop_state_observed']=True
            _FAILURE_STATE['stop_state']='STOPPED'
        else:
            _FAILURE_STATE['restart_running_observed']=True
            _set_phase('restart_recovery')
        _write_private_stop_binding(binding_path)
        _FAILURE_STATE['stop_binding_sha256']=_sha(binding_path)
        _FAILURE_STATE['stop_binding_record_written']=True
    lifecycle._wait_app_state=observed_wait_app_state
    async def observed_admin(profile:str,config_file:str|None,source:dict[str,str],*arguments:str,timeout_seconds:float=120)->str:
        is_stop=len(arguments)>=3 and arguments[0:2]==('apps','stop')
        is_start=len(arguments)>=3 and arguments[0:2]==('apps','start')
        if is_stop:
            if state['stop_id'] is not None: raise RuntimeError('multiple_stop_commands_refused')
            state['stop_id']=uuid.uuid4().hex
            _FAILURE_STATE['stop_attempted']=True
            _FAILURE_STATE.update({
                'stop_correlation_id':state['stop_id'],'deployment_id':deployment_id,
                'source_code_path':source_path,'base_candidate_sha256':base_digest,
                'overlay_snapshot_sha256':overlay_digest,
                'candidate_modules_byte_identical':candidate_bytes_match,
            })
            _write_private_stop_binding(binding_path)
            _set_phase('platform_stop')
        if is_start: _set_phase('restart_start')
        result=await old_admin(profile,config_file,source,*arguments,timeout_seconds=timeout_seconds)
        if is_stop:
            _FAILURE_STATE['stop_command_returned']=True
            _write_private_stop_binding(binding_path)
        if is_start: _set_phase('restart_start_returned')
        return result
    lifecycle._admin_cli=observed_admin
    summary={'schema_version':1,'mode':MODE,'command':'python <private-driver> --mode instrumented-qualification-only --profile <caller-profile> --admin-profile <operator-profile> --settings-file <private-settings> --base-bundle <base-bundle> --overlay-bundle <instrumented-bundle> --deployment-reference <private-ref> --base-candidate-sha256 <digest> --diagnostic-schema <schema> --output-dir <fresh-private-dir>','base_candidate_digest_verified':True,'overlay_snapshot_verified':False,'candidate_modules_byte_identical':candidate_bytes_match,'protected_probe_correlated':False,'durable_row_visible_after_stop':False,'public_qualifier_called':False,'private_receipt_written':False,'status':'failed'}
    try:
        summary['overlay_snapshot_verified']=True
        private_receipt=output/'live-profile-receipt.json'
        _set_phase('public_qualifier')
        with contextlib.redirect_stdout(io.StringIO()), contextlib.redirect_stderr(io.StringIO()):
            await live._run(args.profile,args.admin_profile,args.config_file,source,private_receipt,args.service_caller)
        _set_phase('qualifier_returned')
        summary.update({'public_qualifier_called':True,'protected_probe_correlated':state['probe'] is not None,'durable_row_visible_after_stop':state['row_passed'],'private_receipt_written':private_receipt.is_file(),'status':'passed'})
        (output/'live-profile-receipt.json').chmod(0o600)
        summary['receipt_sha256']=_sha(private_receipt)
        _set_phase('complete')
    finally:
        live.TemporaryDirectory=original_temp
        live.start_log_monitor=original_start
        lifecycle._admin_cli=old_admin
        lifecycle._wait_app_state=old_wait_app_state
    summary.update({key:_FAILURE_STATE.get(key) for key in (
        'phase','stop_attempted','stop_state_observed','row_query_attempted','row_found',
        'reader_status','protected_probe_observed','worker_started_observed',
        'stop_binding_record_written','stop_binding_sha256','restart_running_observed',
    )})
    out=output/'instrumented-result.json';out.write_text(json.dumps(summary,sort_keys=True)+'\n');out.chmod(0o600)
    print(json.dumps(summary,sort_keys=True))
    return 0 if summary['status']=='passed' else 1

        except OSError: pass
        print(json.dumps(safe,sort_keys=True));return 1
