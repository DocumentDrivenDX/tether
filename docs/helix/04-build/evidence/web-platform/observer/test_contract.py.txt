            'command:', '  - "uv"', '  - "run"', '  - "--locked"',
            '  - "python"', '  - "src/app.py"', 'env:',
            '  - name: TETHER_QUALIFICATION_SCENARIO', '    value: "shared-session"',

    async def test_h2c_and_protected_scope_are_nonce_bound(self) -> None:
        nonce = "a" * 32
        nonce_hash = hashlib.sha256(nonce.encode()).hexdigest()
        process_uuid = "b" * 32
        seen: list[dict[str, Any]] = []

        async def app(scope: dict[str, Any], receive: Any, send: Any) -> None:
            del receive
            await send({"type": "http.response.start", "status": 200, "headers": []})
            await send({"type": "http.response.body", "body": b"ok"})
            seen.append(scope)

        proxy = observer._ObservedASGI(app, nonce, process_uuid, nonce_hash, "/api/v1/capabilities")
        async def invoke(path: str, version: str, candidate_nonce: str) -> list[dict[str, Any]]:
            sent: list[dict[str, Any]] = []
            scope = {
                "type": "http", "path": path, "http_version": version,
                "headers": [(b"x-tether-qualification-h2c-probe", candidate_nonce.encode())],
            }
            async def receive() -> dict[str, str]:
                return {"type": "http.request"}
            async def send(message: dict[str, Any]) -> None:
                sent.append(message)
            await proxy(scope, receive, send)
            return sent

        await invoke("/", "2", nonce)
        protected = await invoke("/api/v1/capabilities", "1.1", nonce)
        await invoke("/api/v1/capabilities", "2", "c" * 32)
        self.assertEqual(proxy.observed_version(nonce), "2")
        self.assertEqual(proxy.ingress["asgi_http_version"], "1.1")
        self.assertEqual(proxy.ingress["response_status"], 200)
        self.assertTrue(proxy.ingress["observed"])
        headers = dict(protected[0]["headers"])
        self.assertEqual(headers[b"x-tether-diagnostic-process"], process_uuid.encode())
        self.assertEqual(headers[b"x-tether-diagnostic-nonce-sha256"], nonce_hash.encode())
        self.assertEqual(headers[b"x-tether-diagnostic-asgi-http-version"], b"1.1")
        self.assertEqual(b"".join(message.get("body", b"") for message in protected), b"ok")
        self.assertEqual(len(seen), 3)
